# TensorSharp.Server.Host on Hugging Face Spaces — Docker SDK, CPU-only (GGML CPU backend).
#
# Read the docs: https://huggingface.co/docs/hub/spaces-sdks-docker
#
# Deploy steps (Space: https://huggingface.co/spaces/zhongkaifu/tensorsharp):
#   1. Copy this file into the Space repo as a file named exactly `Dockerfile`.
#   2. Add the YAML header from Dockers/README.md to the Space's README.md
#      (it must contain `sdk: docker`).
#   3. Push. The Space builds the image and starts the server.
#
# The server listens on 0.0.0.0:5000 by default; the runtime stage sets PORT to
# 7860 (the Hugging Face Spaces default port), so the Space needs no `app_port`
# override.

############################################################
# Stage 1 — build the server and the native GGML CPU bridge
############################################################
FROM mcr.microsoft.com/dotnet/sdk:10.0 AS build

# git    : clones TensorSharp and (during the build) ggml-org/ggml into ExternalProjects/ggml
# cmake  : configures/builds the native GGML bridge (TensorSharp.GGML.Native/CMakeLists.txt)
# build-essential : C/C++ toolchain (gcc/g++/make) the CMake build needs
RUN apt-get update \
    && apt-get install -y --no-install-recommends \
        git cmake build-essential ca-certificates \
    && rm -rf /var/lib/apt/lists/*

ENV DOTNET_CLI_TELEMETRY_OPTOUT=1 \
    DOTNET_NOLOGO=1

# Override TENSORSHARP_REF to pin a tag/branch/commit of your choice.
ARG TENSORSHARP_REPO=https://github.com/zhongkaifu/TensorSharp.git
ARG TENSORSHARP_REF=main

WORKDIR /src
RUN git clone --depth 1 --branch ${TENSORSHARP_REF} ${TENSORSHARP_REPO} .

# The app serves the chat UI at "/" itself: Program.cs calls app.UseDefaultFiles(),
# which rewrites "/" to /index.html ahead of the plain-text health route, so the
# Space root (https://<user>-<space>.hf.space) shows the chatbot UI directly. No
# source patch is needed for that.

# Building the server transitively builds Runtime/Models/Backends and runs the
# native GGML CMake build. No nvcc is present, so ggml-cuda is auto-disabled and
# only the CPU backend is compiled (this is what the ggml_cpu backend needs).
# The native build fetches ggml sources over the network on first build.
RUN dotnet build TensorSharp.Server.Host/TensorSharp.Server.Host.csproj -c Release

############################################################
# Stage 2 — runtime image
############################################################
FROM mcr.microsoft.com/dotnet/aspnet:10.0 AS runtime

# libgomp1 : OpenMP runtime the native GGML CPU kernels link against.
# wget     : downloads the GGUF model below.
RUN apt-get update \
    && apt-get install -y --no-install-recommends \
        libgomp1 wget ca-certificates \
    && rm -rf /var/lib/apt/lists/*

# Both the .NET base image and Hugging Face Spaces use UID 1000. The image
# already defines a user at that UID, so reuse it instead of creating one
# (`useradd -m -u 1000` would fail with "UID 1000 is not unique"). Just make the
# working tree (where the server writes uploads/, logs/, and the model lands)
# owned by UID 1000 so every runtime write succeeds.
RUN mkdir -p /home/user/app/models \
    && chown -R 1000:1000 /home/user
USER 1000
ENV HOME=/home/user \
    PATH=/home/user/.local/bin:$PATH
WORKDIR /home/user/app

# The Release build output is a runnable framework-dependent app: managed DLLs,
# the native libGgmlOps.so, runtimes/<rid>/native/* and the wwwroot/ web UI.
COPY --chown=1000:1000 --from=build /src/TensorSharp.Server.Host/bin/ ./

# Download the GGUF model for the demo (gemma-4-E2B QAT Q4_0, ~3.4 GB) plus its
# multimodal projector (mmproj, ~1 GB) so vision input works out of the box.
# Override either build arg to host a different text-only model:
#   docker build \
#     --build-arg MODEL_URL=https://huggingface.co/unsloth/Qwen3.5-9B-GGUF/resolve/main/Qwen3.5-9B-Q8_0.gguf \
#     --build-arg MODEL_FILE=Qwen3.5-9B-Q8_0.gguf ...
ARG MODEL_URL=https://huggingface.co/huihui-ai/Huihui-gemma-4-E2B-it-qat-q4_0-unquantized-abliterated-GGUF/resolve/main/Huihui-gemma-4-E2B-it-qat-q4_0-unquantized-abliterated-Q4_K.gguf
ARG MODEL_FILE=Huihui-gemma-4-E2B-it-qat-q4_0-unquantized-abliterated-Q4_K.gguf
ENV MODEL_FILE=${MODEL_FILE}
RUN wget -q -O "/home/user/app/models/${MODEL_FILE}" "${MODEL_URL}"

# The multimodal projector from the same repo. Set MMPROJ_FILE to an empty
# string at build time to host a text-only server (the CMD drops --mmproj then).
ARG MMPROJ_URL=https://huggingface.co/huihui-ai/Huihui-gemma-4-E2B-it-qat-q4_0-unquantized-abliterated-GGUF/resolve/main/mmproj-model-bf16.gguf
ARG MMPROJ_FILE=mmproj-model-bf16.gguf
ENV MMPROJ_FILE=${MMPROJ_FILE}
RUN if [ -n "${MMPROJ_FILE}" ]; then \
        wget -q -O "/home/user/app/models/${MMPROJ_FILE}" "${MMPROJ_URL}"; \
    fi

# 7860 is the port Hugging Face Spaces routes to by default (no app_port needed).
# The server reads PORT, so this is all it takes to move the listener; override
# it at `docker run` time (-e PORT=8080 -p 8080:8080) for other hosts.
ARG APP_PORT=7860
ENV PORT=${APP_PORT}

ENV DOTNET_CLI_TELEMETRY_OPTOUT=1 \
    DOTNET_NOLOGO=1 \
    BACKEND=ggml_cpu

EXPOSE ${APP_PORT}

# Shell form so the variables expand. Host the model (and its projector, when
# MMPROJ_FILE is set) on the native GGML CPU backend.
CMD dotnet TensorSharp.Server.Host.dll \
    --model "/home/user/app/models/${MODEL_FILE}" \
    ${MMPROJ_FILE:+--mmproj "/home/user/app/models/${MMPROJ_FILE}"} \
    --backend ggml_cpu \
    --max-tokens 4096
