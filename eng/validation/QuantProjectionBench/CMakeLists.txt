cmake_minimum_required(VERSION 3.21)
project(TensorSharpQuantProjectionBench LANGUAGES C CXX)

# This project imports frozen owned CUDA objects and unchanged ggml libraries.
# It never fetches, compiles, or patches an upstream source tree.
set(TSG_GGML_ROOT "" CACHE PATH "Unchanged ggml checkout")
set(TSG_GGML_LIBRARIES "" CACHE PATH "Compatible prebuilt ggml libraries")
set(TSG_PROJECTION_NATIVE_ROOT "" CACHE PATH "Frozen TensorSharp native sources")
set(TSG_PROJECTION_OWNED_CUDA_OBJECTS "" CACHE STRING "Exact four receipt-bound owned CUDA objects")
option(TSG_PROJECTION_ENABLE_CUDA "Enable the resident CUDA benchmark" ON)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use a fresh out-of-source build directory")
endif()
foreach(_directory TSG_GGML_ROOT TSG_GGML_LIBRARIES TSG_PROJECTION_NATIVE_ROOT)
    if(NOT IS_ABSOLUTE "${${_directory}}" OR NOT IS_DIRECTORY "${${_directory}}")
        message(FATAL_ERROR "${_directory} must identify an existing absolute directory")
    endif()
endforeach()
foreach(_file ggml_ops_q8_precision.h ggml_ops_q8_precision.cpp ggml_ops_dsv4_fused.h
        ggml_ops_matmul_precision.h ggml_ops_matmul_precision.cpp)
    if(NOT EXISTS "${TSG_PROJECTION_NATIVE_ROOT}/${_file}")
        message(FATAL_ERROR "Missing frozen owned source: ${_file}")
    endif()
endforeach()
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS C CXX)
set(_libraries ggml ggml-base ggml-cpu)
if(TSG_PROJECTION_ENABLE_CUDA)
    find_package(CUDAToolkit REQUIRED)
    list(APPEND _libraries ggml-cuda)
    list(LENGTH TSG_PROJECTION_OWNED_CUDA_OBJECTS _count)
    if(NOT _count EQUAL 4)
        message(FATAL_ERROR "Exactly four frozen owned CUDA objects are required")
    endif()
    set(_expected ggml_ops_matmul_precision ggml_ops_dsv4_fused ggml_ops_attention_precision ggml_ops_q8_precision)
    foreach(_object IN LISTS TSG_PROJECTION_OWNED_CUDA_OBJECTS)
        if(NOT IS_ABSOLUTE "${_object}" OR NOT EXISTS "${_object}" OR IS_DIRECTORY "${_object}")
            message(FATAL_ERROR "Missing absolute owned object: ${_object}")
        endif()
        get_filename_component(_name "${_object}" NAME)
        if(NOT _name MATCHES "^(ggml_ops_[a-z0-9_]+)\\.cu\\.(obj|o)$")
            message(FATAL_ERROR "Unrecognized owned CUDA object: ${_name}")
        endif()
        list(FIND _expected "${CMAKE_MATCH_1}" _index)
        if(_index LESS 0)
            message(FATAL_ERROR "Unexpected or duplicate owned object: ${_name}")
        endif()
        list(REMOVE_AT _expected ${_index})
    endforeach()
endif()
foreach(_library IN LISTS _libraries)
    if(WIN32)
        set(_file "${TSG_GGML_LIBRARIES}/${_library}.lib")
    else()
        set(_file "${TSG_GGML_LIBRARIES}/lib${_library}.a")
    endif()
    if(NOT EXISTS "${_file}" OR IS_DIRECTORY "${_file}")
        message(FATAL_ERROR "Missing prebuilt library: ${_file}")
    endif()
    add_library(${_library} STATIC IMPORTED GLOBAL)
    set_target_properties(${_library} PROPERTIES IMPORTED_LOCATION "${_file}")
endforeach()
set_target_properties(ggml-base PROPERTIES
    INTERFACE_INCLUDE_DIRECTORIES "${TSG_GGML_ROOT}/include"
    INTERFACE_LINK_LIBRARIES "OpenMP::OpenMP_C;OpenMP::OpenMP_CXX;Threads::Threads")
set_target_properties(ggml-cpu PROPERTIES INTERFACE_LINK_LIBRARIES "ggml-base")
set_target_properties(ggml PROPERTIES INTERFACE_LINK_LIBRARIES "ggml-base;ggml-cpu")

add_executable(QuantProjectionBench quant_projection_bench.cpp
    "${TSG_PROJECTION_NATIVE_ROOT}/ggml_ops_matmul_precision.cpp"
    "${TSG_PROJECTION_NATIVE_ROOT}/ggml_ops_q8_precision.cpp")
target_compile_features(QuantProjectionBench PRIVATE cxx_std_17)
set_target_properties(QuantProjectionBench PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(QuantProjectionBench PRIVATE "${TSG_PROJECTION_NATIVE_ROOT}"
    "${TSG_GGML_ROOT}/include" "${TSG_GGML_ROOT}/src")
target_link_libraries(QuantProjectionBench PRIVATE ggml ggml-cpu ggml-base)
if(MSVC)
    target_compile_options(QuantProjectionBench PRIVATE /EHsc)
endif()
if(TSG_PROJECTION_ENABLE_CUDA)
    set_source_files_properties(${TSG_PROJECTION_OWNED_CUDA_OBJECTS} PROPERTIES EXTERNAL_OBJECT TRUE)
    target_sources(QuantProjectionBench PRIVATE ${TSG_PROJECTION_OWNED_CUDA_OBJECTS})
    target_compile_definitions(QuantProjectionBench PRIVATE TSG_GGML_USE_CUDA=1)
    set_target_properties(ggml-cuda PROPERTIES
        INTERFACE_LINK_LIBRARIES "ggml-base;CUDA::cudart;CUDA::cublas;CUDA::cuda_driver")
    target_link_libraries(QuantProjectionBench PRIVATE ggml-cuda)
endif()
