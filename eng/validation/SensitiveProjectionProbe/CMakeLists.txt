cmake_minimum_required(VERSION 3.21)
project(TensorSharpSensitiveProjectionProbe LANGUAGES C CXX)

# This standalone driver only imports prebuilt ggml libraries. It deliberately
# has no FetchContent, add_subdirectory or upstream source compilation.
set(TSG_GGML_ROOT "" CACHE PATH "Absolute path to the unchanged ggml checkout")
set(TSG_GGML_LIBRARIES "" CACHE PATH "Absolute directory containing the compatible ggml static libraries")
option(TSG_PROJECTION_ENABLE_CUDA "Build the CUDA probe against prebuilt ggml-cuda" ON)
set(TSG_PROJECTION_OWNED_CUDA_OBJECTS "" CACHE STRING
    "Explicit semicolon-separated frozen TensorSharp matmul/fused/attention/q8 CUDA object paths; no CUDA source compilation")
get_filename_component(_repository "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(TSG_PROJECTION_NATIVE_ROOT "${_repository}/TensorSharp.GGML.Native" CACHE PATH
    "TensorSharp native source directory containing the frozen projection fixture and headers")

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use a fresh out-of-source build directory under artifacts/ or another validation cache.")
endif()
foreach(_required TSG_GGML_ROOT TSG_GGML_LIBRARIES TSG_PROJECTION_NATIVE_ROOT)
    if(NOT IS_ABSOLUTE "${${_required}}" OR NOT IS_DIRECTORY "${${_required}}")
        message(FATAL_ERROR "${_required} must explicitly identify an existing absolute directory: '${${_required}}'")
    endif()
endforeach()
foreach(_header include/ggml.h include/ggml-backend.h include/ggml-cpu.h src/ggml-impl.h)
    if(NOT EXISTS "${TSG_GGML_ROOT}/${_header}")
        message(FATAL_ERROR "Missing unchanged ggml header: ${TSG_GGML_ROOT}/${_header}")
    endif()
endforeach()
foreach(_source tests/precision_test_utils.h ggml_ops_precision_policy.h ggml_ops_matmul_precision.h
        ggml_ops_matmul_precision.cpp ggml_ops_q8_precision.h ggml_ops_q8_precision.cpp ggml_ops_dsv4_fused.h)
    if(NOT EXISTS "${TSG_PROJECTION_NATIVE_ROOT}/${_source}")
        message(FATAL_ERROR "Missing frozen TensorSharp fixture dependency: ${TSG_PROJECTION_NATIVE_ROOT}/${_source}")
    endif()
endforeach()

find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS C CXX)
set(_imports ggml ggml-base ggml-cpu)
if(TSG_PROJECTION_ENABLE_CUDA)
    find_package(CUDAToolkit REQUIRED)
    if(NOT EXISTS "${TSG_GGML_ROOT}/include/ggml-cuda.h")
        message(FATAL_ERROR "Missing unchanged ggml CUDA header: ${TSG_GGML_ROOT}/include/ggml-cuda.h")
    endif()
    list(APPEND _imports ggml-cuda)
    list(LENGTH TSG_PROJECTION_OWNED_CUDA_OBJECTS _object_count)
    if(NOT _object_count EQUAL 4)
        message(FATAL_ERROR "CUDA mode requires exactly four frozen owned CUDA objects in TSG_PROJECTION_OWNED_CUDA_OBJECTS")
    endif()
    set(_expected_objects ggml_ops_matmul_precision ggml_ops_dsv4_fused
        ggml_ops_attention_precision ggml_ops_q8_precision)
    foreach(_object IN LISTS TSG_PROJECTION_OWNED_CUDA_OBJECTS)
        if(NOT IS_ABSOLUTE "${_object}" OR NOT EXISTS "${_object}" OR IS_DIRECTORY "${_object}")
            message(FATAL_ERROR "Owned CUDA object must be an existing absolute file: ${_object}")
        endif()
        get_filename_component(_name "${_object}" NAME)
        if(NOT _name MATCHES "^(ggml_ops_[a-z0-9_]+)\\.cu\\.(obj|o)$")
            message(FATAL_ERROR "Unrecognized owned CUDA object name: ${_name}")
        endif()
        list(FIND _expected_objects "${CMAKE_MATCH_1}" _index)
        if(_index LESS 0)
            message(FATAL_ERROR "Unexpected or duplicate owned CUDA object: ${_name}")
        endif()
        list(REMOVE_AT _expected_objects ${_index})
    endforeach()
endif()
foreach(_library IN LISTS _imports)
    if(WIN32)
        set(_path "${TSG_GGML_LIBRARIES}/${_library}.lib")
    else()
        set(_path "${TSG_GGML_LIBRARIES}/lib${_library}.a")
    endif()
    if(NOT EXISTS "${_path}" OR IS_DIRECTORY "${_path}")
        message(FATAL_ERROR "Missing required prebuilt static library: ${_path}")
    endif()
    add_library(${_library} STATIC IMPORTED GLOBAL)
    set_target_properties(${_library} PROPERTIES IMPORTED_LOCATION "${_path}" IMPORTED_NO_SYSTEM TRUE)
endforeach()
set_target_properties(ggml-base PROPERTIES
    INTERFACE_INCLUDE_DIRECTORIES "${TSG_GGML_ROOT}/include"
    INTERFACE_LINK_LIBRARIES "OpenMP::OpenMP_C;OpenMP::OpenMP_CXX;Threads::Threads")
set_target_properties(ggml-cpu PROPERTIES INTERFACE_LINK_LIBRARIES "ggml-base")
set_target_properties(ggml PROPERTIES INTERFACE_LINK_LIBRARIES "ggml-base;ggml-cpu"
    INTERFACE_COMPILE_DEFINITIONS "GGML_USE_CPU")
if(TSG_PROJECTION_ENABLE_CUDA)
    set_target_properties(ggml-cuda PROPERTIES
        INTERFACE_LINK_LIBRARIES "ggml-base;CUDA::cudart;CUDA::cublas;CUDA::cuda_driver")
    set_property(TARGET ggml APPEND PROPERTY INTERFACE_LINK_LIBRARIES ggml-cuda)
    set_property(TARGET ggml APPEND PROPERTY INTERFACE_COMPILE_DEFINITIONS GGML_USE_CUDA)
endif()

function(add_projection_probe target fixture)
    add_executable(${target} "${CMAKE_CURRENT_LIST_DIR}/${fixture}"
        "${TSG_PROJECTION_NATIVE_ROOT}/ggml_ops_matmul_precision.cpp")
    if(fixture STREQUAL "sensitive_q8_projection_test.cpp" OR fixture STREQUAL "sensitive_q5_projection_test.cpp")
        target_sources(${target} PRIVATE "${TSG_PROJECTION_NATIVE_ROOT}/ggml_ops_q8_precision.cpp")
    endif()
    target_compile_features(${target} PRIVATE cxx_std_17)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_include_directories(${target} PRIVATE "${TSG_PROJECTION_NATIVE_ROOT}"
        "${TSG_PROJECTION_NATIVE_ROOT}/tests" "${TSG_GGML_ROOT}/include" "${TSG_GGML_ROOT}/src")
    if(MSVC)
        target_compile_options(${target} PRIVATE /EHsc)
    endif()
endfunction()
